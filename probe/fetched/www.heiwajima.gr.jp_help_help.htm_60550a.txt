      document.addEventListener("DOMContentLoaded", () => {
        const tabs = document.querySelectorAll(".tab_bt input");
        const tabLabels = document.querySelectorAll(".tab_bt label");
        const tabContents = document.querySelectorAll(".tab_content");

        tabs.forEach((radio) => {
          radio.addEventListener("change", () => {
            // すべてのタブラベルから .active を削除
            tabLabels.forEach((label) => label.classList.remove("active"));

            // すべてのタブコンテンツを非表示にする
            tabContents.forEach((content) => content.classList.remove("active"));

            // 選択されたラジオボタンに対応するタブラベルに .active を追加
            const checkedRadioId = radio.id;
            const checkedLabel = document.querySelector(`label[for="${checkedRadioId}"]`);
            if (checkedLabel) {
              checkedLabel.classList.add("active");
            }

            // 選択されたラジオボタンに対応するタブコンテンツを表示
            const targetId = `${checkedRadioId}_content`;
            const targetContent = document.getElementById(targetId);
            if (targetContent) {
              targetContent.classList.add("active");
            }
          });
        });

        // 初期表示設定（最初のタブを表示）
        document.querySelector('input[name="tab_item"]:checked').dispatchEvent(new Event("change"));
      });
